The projection graph is created with

export function createProjectionGraph(graph: Graph, view: ViewI) {
    const node_to_cluster_map = getNodeClusterMap(graph, view.clusters);
    // find which nodes to render
    const renderIds = new Set<string>();
    graph.forEachNode((nodeId) => {
        const render_id = node_to_cluster_map.get(nodeId);
        if (render_id) renderIds.add(render_id);
    });

    const proj_graph = new DirectedGraph<NodeAttributesI>({});

    // add base nodes
    graph.forEachNode((node_id, attributes) => {
        if (renderIds.has(node_id)) {
            proj_graph.addNode(node_id, attributes);
        }
    });

    // add cluster nodes
    for (const cluster of view.clusters.values()) {
        // expanded clusters not rendered
        if (cluster.expanded) continue;

        renderIds.add(cluster.id);

        const attr: NodeAttributesI = {
            id: cluster.id,
            label: cluster.label,
            type: "cluster",
            full_path: "",
            external: false,
        };
        proj_graph.addNode(cluster.id, attr);
    }

    graph.forEachEdge((edgeKey, attrs, source, target) => {
        const from = node_to_cluster_map.get(source);
        const to = node_to_cluster_map.get(target);

        if (!proj_graph.hasEdge(from, to) && from !== to) {
            proj_graph.addEdge(from, to);
        }
    });

    return proj_graph;
}
